Circular Palindromes
https://www.hackerrank.com/challenges/circular-palindromes/problem
Program:-
import java.util.*;

class Solution {

    static int longest(String s) {
        int best = 1;

        for (int i = 0; i < s.length(); i++) {
            best = Math.max(best, expand(s, i, i));
            best = Math.max(best, expand(s, i, i + 1));
        }

        return best;
    }

    static int expand(String s, int l, int r) {
        while (l >= 0 && r < s.length()
                && s.charAt(l) == s.charAt(r)) {
            l--;
            r++;
        }
        return r - l - 1;
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        int n = sc.nextInt();
        String s = sc.next();

        for (int k = 0; k < n; k++) {
            String rotation = s.substring(k) + s.substring(0, k);
            System.out.println(longest(rotation));
        }
    }
}
